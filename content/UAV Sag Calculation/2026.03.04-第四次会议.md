% ========== 定义jisuan函数 ==========
function [max_diff, a, b, diff_list] = jisuan(w_point)
    % JISUAN 基于w_point的第一个点和最后一个点拟合直线，计算所有点y值与直线预测值的最大差值
    % 输入：%   w_point - N×3的矩阵，每一行是[x,y,z]坐标
    % 输出：%   max_diff - 最大差值（绝对值）
    %   a - 直线斜率   b - 直线截距
    %   diff_list - 每个点的差值列表（y1 - y0，y1是预测值，y0是实际值）
    % 提取x、y坐标（忽略z）
    x = w_point(:,1);
    y0 = w_point(:,2);
    % 取第一个点和最后一个点拟合直线 y = ax + b 【核心修改点】
    x1 = x(1);         % 第一个点的x坐标
    y1_0 = y0(1);      % 第一个点的y坐标
    x2 = x(end);       % 最后一个点的x坐标
    y2_0 = y0(end);    % 最后一个点的y坐标
    % 计算斜率和截距（避免分母为0）
    if x2 - x1 ~= 0
        a = (y2_0 - y1_0) / (x2 - x1);  % 斜率
        b = y1_0 - a * x1;              % 截距
    else
        % 垂直直线（x固定），特殊处理
        a = inf;
        b = x1;
        disp(['警告：第一个点和最后一个点x坐标相同，拟合为垂直直线 x = ', num2str(b)]);
    end
    % 计算每个点的预测y值和差值
    diff_list = zeros(size(y0));
    for i = 1:length(x)
        if a ~= inf
            y1 = a * x(i) + b;  % 直线上的预测y值
            diff_list(i) = y1 - y0(i);  % 预测值 - 实际值
        else
            % 垂直直线时，差值为x方向偏差（x_i - x1）
            diff_list(i) = x(i) - b;
        end
    end
    % 计算最大差值（绝对值）
    max_diff = max(abs(diff_list));
end

## 折算fx、fy
### 一、
$$depth = \frac{f \cdot \text{baseline}}{disp} = \frac{f_x \cdot B}{d}$$
$$其中 depth为深度，即Z，单位mm；f为焦距，单位pixel$$
$$baseline为基线长度（左右光心距离），单位mm$$
$$disp为视差，同一个三维点在左、右图像上的横坐标像素差，disp=x_l-x_r=(u_l-cx_l)-(u_r-cx_r)，单位pixel$$
该公式的前提是图像进行<font color="#d83931">极线校正</font>之后，即$v_l=v_r$

### 二、采用能读取焦距的相机
计算传感器尺寸
$$\text{Sensor Size} = 2 \times f \times \tan\left(\frac{\text{FOV}}{2}\right)$$
$$其中 \text{Sensor Size}为传感器尺寸，单位mm；f为焦距，单位mm；FOV为水平视野，单位°$$
计算像素焦距
$$\text{Pixel Focal Length} = \frac{\text{Focal Length in mm}}{\text{Sensor Size in mm}} \times \text{Image Size in pixels}$$
例：假设水平视野FOV是40°，镜头焦距8mm，图像大小是1920·1200
则$$\text{Sensor Size} = 2 \times f \times \tan\left(\frac{\text{FOV}}{2}\right)=2 \times 8 \times \tan\left(\frac{\text{40°}}{2}\right)≈5.76 mm$$
水平方向像素焦距fx为：
$$f_x=\text{Pixel Focal Length} = \frac{\text{8}}{\text{5.76}} \times \text{1920}≈2667pixels$$
垂直方向像素焦距fy为：
$$f_y=\text{Pixel Focal Length} = \frac{\text{8}}{\text{5.76}\times\text{1200/1920}} \times \text{1200}≈2667pixels$$

### 三、
获取输电塔的gps信息和无人机的gps信息，将输电塔点S的世界坐标转换为以左相机光心为原点的相机坐标系，此时点S坐标为$P_c=(X_c,Y_c,Z_c)$，然后拍摄的左相机图像上选取S点的像素坐标$(u_l,v_l)$值
根据以下公式
左相机
$$fx_l=(u_l-cx_l)\times\frac{Z_c}{X_c}$$
$$fy_l=(v_l-cy_l)\times\frac{Z_c}{Y_c}$$
右相机，以右相机光心建立的坐标系下S点坐标为$P_c=RP_{cl}+T$
$$fx_r=(u_r-cx_r)\times\frac{Z_c}{X_c}$$
$$fy_r=(v_r-cy_r)\times\frac{Z_c}{Y_c}$$
